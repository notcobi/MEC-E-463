\section{Utility}
\subsection{Common Unit Conversions}
\begin{itemize}
    \item $1~\text{atm} = 33.9~\text{ft wg}$
    \item $1~\text{gal} = \frac{1}{7.48}~\text{ft}^3$
    \item $1~\text{gpm} = \frac{1}{448.8}~\text{ft}^3/\text{s}$
    \item $1~\text{cfm} = 0.0004719~\text{m}^3/\text{s}$
    \item $g = 9.81~\text{m/s}^2 = 32.2~\text{ft/s}^2$
    \item $1~\text{J/kg} = 1~\text{m}^2/\text{s}^2$
    \item $1~\text{psi} = 6895~\text{Pa}$
    \item $1~\text{in. wg} = 249.1~\text{Pa}$ at 70°F
    \item $1~\text{psi} = 27.68~\text{in. wg}$
    \item $1~\text{ft} = 12~\text{in}$, \quad $1~\text{m} = 3.281~\text{ft}$
\end{itemize}
